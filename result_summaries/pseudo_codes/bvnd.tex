\begin{algorithm}[H]
\caption{Basic Variable Neighborhood Descent (BVND)}
\begin{algorithmic}[1]
\Require Initial solution $S$, neighborhood list $\mathcal{N}$
\Ensure Improved solution $S$

\State $no\_improve \gets 0$

\While{$no\_improve < \text{max\_no\_improve}$}
    \State $improved \gets \text{false}$

    \For{each neighborhood $N_k \in \mathcal{N}$}
        \State $S' \gets \text{Explore}(S, N_k)$

        \If{$S'$ improves $S$}
            \State $S \gets S'$
            \State $improved \gets \text{true}$
            \State \textbf{break} \Comment{restart from first neighborhood}
        \EndIf
    \EndFor

    \If{$improved$}
        \State $no\_improve \gets 0$
    \Else
        \State $no\_improve \gets no\_improve + 1$
    \EndIf
\EndWhile

\State \Return $S$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Neighborhood Exploration}
\begin{algorithmic}[1]
\Require Solution $S$, neighborhood $N_k$
\Ensure Candidate solution $S'$

\For{each move $m \in N_k(S)$}
    \State Apply move $m$ to obtain $S_{\text{cand}}$

    \If{first-improvement strategy}
        \If{$S_{\text{cand}}$ improves $S$}
            \State \Return $S_{\text{cand}}$
        \EndIf
    \Else
        \State Keep the best candidate found so far
    \EndIf
\EndFor

\State \Return best candidate (if any)
\end{algorithmic}
\end{algorithm}
