\begin{algorithm}[H]
\caption{RF-Integrated Neighborhood Exploration}
\begin{algorithmic}[1]
\Require Solution $S$, neighborhood $N_k$, model, scaler, state\_extractor
\Ensure Candidate solution $S'$

\State Extract state features: $\mathbf{state\_features} \gets \text{state\_extractor.extract}(S, N_k)$
\State \phantom{}

\State \# \text{State duplication}
\State $\mathbf{state\_FI} \gets \text{append}(\mathbf{state\_features}, 0)$
\State $\mathbf{state\_BI} \gets \text{append}(\mathbf{state\_features}, 1)$
\State \phantom{}

\State \# \text{State normalization}
\State $\mathbf{state\_FI\_norm} \gets \text{scaler.transform}(\mathbf{state\_FI}.\text{reshape}(1, -1))$
\State $\mathbf{state\_BI\_norm} \gets \text{scaler.transform}(\mathbf{state\_BI}.\text{reshape}(1, -1))$
\State \phantom{}

\State \# \text{FI and BI predictions}
\State $p_{\text{FI}} \gets \text{model.classifier.predict\_proba}(\mathbf{state\_FI\_norm})[0, 1]$
\State $g_{\text{FI}} \gets \text{model.regressor.predict}(\mathbf{state\_FI\_norm})[0]$
\State \phantom{}
\State $p_{\text{BI}} \gets \text{model.classifier.predict\_proba}(\mathbf{state\_BI\_norm})[0, 1]$
\State $g_{\text{BI}} \gets \text{model.regressor.predict}(\mathbf{state\_BI\_norm})[0]$
\State \phantom{}

\State \# \text{Score computation and decision}
\State $\text{score}_{\text{FI}} \gets p_{\text{FI}} \cdot \max(g_{\text{FI}}, 0)$
\State $\text{score}_{\text{BI}} \gets p_{\text{BI}} \cdot \max(g_{\text{BI}}, 0)$
\State \phantom{}
\If{$\max(\text{score}_{\text{FI}}, \text{score}_{\text{BI}}) < 0.01$}
    \State $\text{use\_best\_improvement} \gets \text{false}$
\Else
    \State $\text{use\_best\_improvement} \gets (\text{score}_{\text{BI}} > \text{score}_{\text{FI}})$
\EndIf
\State \phantom{}

\State \# \text{Apply neighborhood with chosen strategy}
\State $new\_solution \gets \text{apply\_neighbourhood}(current\_solution, neighbourhood\_id, first\_improve=\neg use\_best\_improvement)$
\State \Return $new\_solution$
\end{algorithmic}
\end{algorithm}
